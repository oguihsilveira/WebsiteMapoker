create table funcionarios(
codigo INT(5) not null,
nome VARCHAR(100) not null,
email VARCHAR(100) not null,
datanasc DATE not null,
cargo VARCHAR(50) not null,
salario FLOAT(10,2) not null,
endereco VARCHAR(100) not null,
carga_horaria VARCHAR(100) not null,
PRIMARY KEY (codigo)
);
-
create table usuarios(
codigo INT (5) not null,
login VARCHAR(50) not null,
senha VARCHAR(12) not null,
cod_funcionario INT(5) not null,
PRIMARY KEY (codigo),
FOREIGN KEY (cod_funcionario) REFERENCES funcionarios(codigo)
);
-
create table estoque (
codigo INT(5) PRIMARY KEY NOT NULL,
item VARCHAR(50) NOT NULL,
tipo VARCHAR(50) NOT NULL,
observacoes VARCHAR(50) NOT NULL,
preco_compra FLOAT(10, 2) NOT NULL,
preco_venda FLOAT(10, 2) NOT NULL,
data_entrada DATETIME NOT NULL,
qtde_entrada INT(5) NOT NULL,
cod_funcionario INT(5) NOT NULL,
FOREIGN KEY (cod_funcionario) REFERENCES funcionarios(codigo)
);
-
create table produtos (
codigo INT(5) PRIMARY KEY, 
item VARCHAR(50) NOT NULL, 
tipo VARCHAR(50) NOT NULL,
preco_padrao FLOAT(10, 2) NOT NULL, 
desconto FLOAT(10, 2), 
status VARCHAR(50) NOT NULL, 
quantidade INT(5) NOT NULL, 
foto VARCHAR(255) NOT NULL, 
observacoes VARCHAR(50) NOT NULL, 
cod_estoque INT(5) NOT NULL, 
FOREIGN KEY (cod_estoque) REFERENCES estoque(codigo)
);
-
create table clientes (
codigo INT(5) PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(50) NOT NULL,
empresa VARCHAR(50) NOT NULL,
telefone VARCHAR(50) NOT NULL,
email VARCHAR(50) NOT NULL,
login VARCHAR(50) NOT NULL,
senha VARCHAR(255) NOT NULL
);
-
create table pedidos (
codigo INT(5) PRIMARY KEY AUTO_INCREMENT,
item VARCHAR(100) NOT NULL,
quantidade INT(5) NOT NULL,
destinatario VARCHAR(100) NOT NULL,
endereco VARCHAR(100) NOT NULL,
tipo_pgto VARCHAR(50) NOT NULL,
qntd_parcelas INT(2) NOT NULL,
data_compra DATE NOT NULL,
valor_compra FLOAT(10, 2) NOT NULL,
valor_parcela FLOAT(10, 2) NOT NULL,
status VARCHAR(50) NOT NULL,
cod_produto INT(5) NOT NULL,
cod_cliente INT(5) NOT NULL,
FOREIGN KEY (cod_produto) REFERENCES produtos(codigo),
FOREIGN KEY (cod_cliente) REFERENCES clientes(codigo)
);
